40fn clear_bss() {
41    unsafe extern "C" {
42        safe fn sbss();
43        safe fn ebss();
44    }
45    unsafe {
46        core::slice::from_raw_parts_mut(
47            linker_symbol_addr!(sbss) as *mut u8,
48            linker_symbol_addr!(ebss) - linker_symbol_addr!(sbss),
49        )
50        .fill(0);
51    }
52}
