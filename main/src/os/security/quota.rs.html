3use super::{IpcError, IpcOperation, IpcRequest, IpcResult};

5/// Maximum number of open file descriptors per process.
6pub const MAX_OPEN_FILES: usize = 32;
7
8/// Maximum number of pipe endpoint file descriptors per process.
9///
10/// One pipe created by `pipe()` normally consumes two pipe FDs:
11/// one readable endpoint and one writable endpoint.
12pub const MAX_OPEN_PIPE_FDS: usize = 16;
13
14/// A successful pipe creation installs one read and one write endpoint.
15const PIPE_ENDPOINTS_PER_PIPE: usize = 2;
16
17/// Per-process IPC resource accounting state.
18#[derive(Clone, Copy, Debug, Eq, PartialEq)]
19pub struct QuotaState {
20    /// Number of currently open file descriptors.
21    ///
22    /// This includes ordinary files, stdio descriptors, and pipe endpoints.
23    pub open_files: usize,
24
25    /// Number of currently open pipe endpoint file descriptors.
26    ///
27    /// This counts endpoints rather than pipe objects. Therefore one newly
28    /// created pipe normally contributes two.
29    pub open_pipes: usize,
30
31    /// Bitmap recording which descriptor numbers refer to pipe endpoints.
32    ///
33    /// `MAX_OPEN_FILES` is currently 32, so a u64 is sufficient.
34    pipe_fd_mask: u64,
35}
36
37impl Default for QuotaState {
38    fn default() -> Self {
39        Self::initial()
40    }
41}
42
43impl QuotaState {
44    /// Initial quota state.
45    ///
46    /// The initial process starts with stdin, stdout, and stderr.
47    pub const fn initial() -> Self {
48        Self {
49            open_files: 3,
50            open_pipes: 0,
51            pipe_fd_mask: 0,
52        }
53    }
54
55    /// Fork inherits the parent's descriptor table, so the child starts with
56    /// the same per-process usage counters and pipe-FD metadata.
57    pub const fn fork_from(parent: &Self) -> Self {
58        *parent
59    }
60
61    /// Reserve ordinary file descriptors.
62    ///
63    /// On failure the state is left unchanged.
64    pub fn reserve_files(&mut self, amount: usize) -> IpcResult<()> {
65        let next = self
66            .open_files
67            .checked_add(amount)
68            .ok_or(IpcError::TooManyFiles)?;
69
70        if next > MAX_OPEN_FILES {
71            return Err(IpcError::TooManyFiles);
72        }
73
74        self.open_files = next;
75        Ok(())
76    }
77
78    /// Roll back previously reserved ordinary file descriptors.
79    pub fn release_files(&mut self, amount: usize) {
80        debug_assert!(self.open_files >= amount);
81        self.open_files = self.open_files.saturating_sub(amount);
82    }
83
84    /// Atomically reserve pipe endpoint descriptors.
85    ///
86    /// Every pipe endpoint is also an open file descriptor, so this checks
87    /// both the total FD limit and the pipe-specific limit before changing
88    /// either counter.
89    ///
90    /// If either check fails, neither counter is modified.
91    pub fn reserve_pipe_fds(&mut self, amount: usize) -> IpcResult<()> {
92        let next_files = self
93            .open_files
94            .checked_add(amount)
95            .ok_or(IpcError::TooManyFiles)?;
96
97        if next_files > MAX_OPEN_FILES {
98            return Err(IpcError::TooManyFiles);
99        }
100
101        let next_pipes = self
102            .open_pipes
103            .checked_add(amount)
104            .ok_or(IpcError::ResourceExhausted)?;
105
106        if next_pipes > MAX_OPEN_PIPE_FDS {
107            return Err(IpcError::ResourceExhausted);
108        }
109
110        self.open_files = next_files;
111        self.open_pipes = next_pipes;
112
113        Ok(())
114    }
115
116    /// Roll back pipe descriptors before descriptor numbers are committed.
117    pub fn release_pipe_fds(&mut self, amount: usize) {
118        debug_assert!(self.open_files >= amount);
119        debug_assert!(self.open_pipes >= amount);
120
121        self.open_files = self.open_files.saturating_sub(amount);
122        self.open_pipes = self.open_pipes.saturating_sub(amount);
123    }
124
125    /// Record that an already-reserved descriptor number is a pipe endpoint.
126    ///
127    /// This does not change counters; `reserve_pipe_fds` must have succeeded
128    /// before this method is called.
129    pub fn register_pipe_fd(&mut self, fd: usize) {
130        debug_assert!(fd < MAX_OPEN_FILES);
131        debug_assert!(fd < u64::BITS as usize);
132
133        if fd < u64::BITS as usize {
134            self.pipe_fd_mask |= 1u64 << fd;
135        }
136    }
137
138    /// Return whether a descriptor number is currently recorded as a pipe.
139    pub fn is_pipe_fd(&self, fd: usize) -> bool {
140        if fd >= u64::BITS as usize {
141            return false;
142        }
143
144        self.pipe_fd_mask & (1u64 << fd) != 0
145    }
146
147    /// Reserve quota for duplicating an existing descriptor.
148    ///
149    /// Returns whether the source descriptor is a pipe endpoint so the
150    /// caller can register the newly allocated descriptor accordingly.
151    pub fn reserve_dup_fd(&mut self, source_fd: usize) -> IpcResult<bool> {
152        let is_pipe = self.is_pipe_fd(source_fd);
153
154        if is_pipe {
155            self.reserve_pipe_fds(1)?;
156        } else {
157            self.reserve_files(1)?;
158        }
159
160        Ok(is_pipe)
161    }
162
163    /// Remove pipe metadata without changing quota counters.
164    ///
165    /// This is used when an operation has installed pipe descriptors but
166    /// later fails. The quota reservation itself is rolled back by
167    /// `complete`.
168    pub fn unregister_pipe_fd(&mut self, fd: usize) {
169        if fd < u64::BITS as usize {
170            self.pipe_fd_mask &= !(1u64 << fd);
171        }
172    }
173
174    /// Release one committed descriptor.
175    ///
176    /// If the descriptor is a pipe endpoint, both the total FD counter and
177    /// the pipe-specific counter are decremented.
178    pub fn release_fd(&mut self, fd: usize) {
179        let was_pipe = self.is_pipe_fd(fd);
180
181        if fd < u64::BITS as usize {
182            self.pipe_fd_mask &= !(1u64 << fd);
183        }
184
185        debug_assert!(self.open_files > 0);
186        self.open_files = self.open_files.saturating_sub(1);
187
188        if was_pipe {
189            debug_assert!(self.open_pipes > 0);
190            self.open_pipes = self.open_pipes.saturating_sub(1);
191        }
192    }
193}

195/// Kind of resource represented by a pending reservation.
196#[derive(Clone, Copy, Debug, Eq, PartialEq)]
197enum QuotaReservationKind {
198    /// This IPC operation does not reserve quota.
199    None,
200
201    /// Pipe endpoint descriptors were reserved.
202    PipeFds,
203}
204
205/// Opaque reservation carried by the security facade.
206///
207/// A successful `preflight` may already have charged quota. `finish` keeps
208/// that charge on success and rolls it back on failure.
209pub(crate) struct QuotaReservation {
210    kind: QuotaReservationKind,
211    amount: usize,
212}
213
214/// Reserve resources requested through the security facade.
215///
216/// Reservation happens before the actual operation. On failure this function
217/// leaves the quota state unchanged.
218pub(crate) fn reserve(state: &mut QuotaState, request: &IpcRequest) -> IpcResult<QuotaReservation> {
219    match request.operation {
220        IpcOperation::PipeCreate => {
221            // IpcRequest.amount follows the audit ABI and counts one pipe
222            // creation operation. Quota accounting charges its two concrete
223            // endpoint descriptors independently of that public value.
224            state.reserve_pipe_fds(PIPE_ENDPOINTS_PER_PIPE)?;
225            Ok(QuotaReservation {
226                kind: QuotaReservationKind::PipeFds,
227                amount: PIPE_ENDPOINTS_PER_PIPE,
228            })
229        }
230        _ => Ok(QuotaReservation {
231            kind: QuotaReservationKind::None,
232            amount: 0,
233        }),
234    }
235}
236
237/// Commit or roll back a quota reservation.
238///
239/// Successful operations keep the quota charged. Failed operations release
240/// exactly the resources reserved during `preflight`.
241pub(crate) fn finish(state: &mut QuotaState, reservation: QuotaReservation, success: bool) {
242    if success {
243        return;
244    }
245
246    match reservation.kind {
247        QuotaReservationKind::None => {}
248        QuotaReservationKind::PipeFds => {
249            state.release_pipe_fds(reservation.amount);
250        }
251    }
