34/// 复制给用户态的稳定记录，字段顺序和类型必须与 ABI v1 一致。
35#[repr(C)]
36#[derive(Clone, Copy, Debug, Default, Eq, PartialEq)]
37pub(crate) struct AuditRecordV1 {
38    pub(crate) abi_version: u16,      // ABI 版本，固定为 1
39    pub(crate) record_size: u16,      // 记录字节数，固定为 80
40    pub(crate) operation: u16,        // 稳定操作编号，不是 Rust 枚举判别值
41    pub(crate) flags: u16,            // 本次读取相关的记录标志
42    pub(crate) errno: i32,            // 成功为 0，失败为正 errno
43    pub(crate) subject_uid: u32,      // 发起请求的 UID
44    pub(crate) object_owner_uid: u32, // 资源所有者 UID，未知时为 u32::MAX
45    pub(crate) reserved0: u32,        // 保留字段，输出时必须为 0
46    pub(crate) sequence: u64,         // 本次启动内的事件序号
47    pub(crate) timestamp_ms: u64,     // 内核时钟毫秒数，不是 Unix 时间
48    pub(crate) subject_pid: u64,      // 发起请求的 PID
49    pub(crate) object_id: u64,        // 稳定资源编号，禁止填写内核地址
50    pub(crate) requested_amount: u64, // 请求的字节数或资源数量
51    pub(crate) result_value: u64,     // 成功时的实际数量，失败时为 0
52    pub(crate) reserved1: u64,        // 保留字段，输出时必须为 0
53}

77/// 将公共操作类型转换为稳定编号；枚举新增变体时必须补充此映射。
78const fn operation_to_audit_id(operation: IpcOperation) -> u16 {
79    match operation {
80        IpcOperation::SignalSend => AUDIT_OP_SIGNAL_SEND,
81        IpcOperation::PipeCreate => AUDIT_OP_PIPE_CREATE,
82        IpcOperation::PipeRead => AUDIT_OP_PIPE_READ,
83        IpcOperation::PipeWrite => AUDIT_OP_PIPE_WRITE,
84        IpcOperation::AuditRead => AUDIT_OP_AUDIT_READ,
85    }
86}

88/// 返回正 errno；系统调用层返回错误时再转换为负 isize。
89pub(crate) const fn error_to_errno(error: IpcError) -> i32 {
90    match error {
91        IpcError::PermissionDenied => 1,   // EPERM：权限不足
92        IpcError::ProcessNotFound => 3,    // ESRCH：进程不存在
93        IpcError::TryAgain => 11,          // EAGAIN：稍后重试
94        IpcError::InvalidAddress => 14,    // EFAULT：用户地址无效
95        IpcError::InvalidArgument => 22,   // EINVAL：参数无效
96        IpcError::TooManyFiles => 24,      // EMFILE：文件描述符数量超限
97        IpcError::ResourceExhausted => 28, // ENOSPC：资源耗尽
98    }
99}

108/// 只在内核中保存的事件，不直接复制给用户态。
109///
110/// 空槽和待写入事件的 sequence 为 0；有效槽由 head 和 len 判定，
111/// 正式事件序号统一在 AuditRing::push 中分配。
112#[derive(Clone, Copy, Debug, Eq, PartialEq)]
113pub(crate) struct AuditEvent {
114    pub(crate) sequence: u64,         // 第几条事件
115    pub(crate) timestamp_ms: u64,     // 发生时间
116    pub(crate) operation: u16,        // 操作类型
117    pub(crate) errno: i32,            // 失败原因，成功为 0
118    pub(crate) subject_pid: u64,      // 发起请求的进程 ID
119    pub(crate) subject_uid: u32,      // 发起请求的用户 ID
120    pub(crate) object_id: u64,        // 请求的对象 ID
121    pub(crate) object_owner_uid: u32, // 请求对象的所有者 ID
122    pub(crate) requested_amount: u64, // 请求的资源量
123    pub(crate) result_value: u64,     // 请求的结果值
124}

126impl AuditEvent {
127    /// 用于初始化环形数组和快照数组的空值，不代表一条已记录的事件。
128    pub(crate) const EMPTY: Self = Self {
129        sequence: 0,
130        timestamp_ms: 0,
131        operation: AUDIT_OP_UNSPECIFIED,
132        errno: 0,
133        subject_pid: 0,
134        subject_uid: 0,
135        object_id: 0,
136        object_owner_uid: 0,
137        requested_amount: 0,
138        result_value: 0,
139    };
140
141    /// 纯数据转换：时间戳由调用者传入，便于独立测试和限制借用范围。
142    ///
143    /// 调用方负责提供稳定资源 ID、未知所有者哨兵，以及符合操作语义的
144    /// 请求数量。当前 RV64 目标的 usize 转为 u64 不会丢失信息。
145    pub(crate) const fn from_request(
146        request: &IpcRequest,
147        outcome: &IpcResult<usize>,
148        timestamp_ms: u64,
149    ) -> Self {
150        let (errno, result_value) = match *outcome {
151            Ok(value) => {
152                // 信号发送和管道创建成功表示完成 1 次操作；读写保存实际字节数。
153                let amount = match request.operation {
154                    IpcOperation::SignalSend | IpcOperation::PipeCreate => 1,
155                    _ => value as u64,
156                };
157                (0, amount)
158            }
159            // 错误码只进入 errno，不能混入成功数量或转换成巨大的无符号数。
160            Err(error) => (error_to_errno(error), 0),
161        };
162
163        Self {
164            sequence: 0,
165            timestamp_ms,
166            operation: operation_to_audit_id(request.operation),
167            errno,
168            subject_pid: request.subject.pid as u64,
169            subject_uid: request.subject.uid,
170            object_id: request.object.id,
171            object_owner_uid: request.object.owner_uid,
172            requested_amount: request.amount as u64,
173            result_value,
174        }
175    }
176
177    /// 构造审计控制调用的失败事件，统计查询的请求数量固定为 0。
178    const fn from_control_failure(
179        operation: AuditControlOperation,
180        subject: &IpcSubject,
181        requested_amount: usize,
182        error: IpcError,
183        timestamp_ms: u64,
184    ) -> Self {
185        let (operation, requested_amount) = match operation {
186            AuditControlOperation::AuditRead => (AUDIT_OP_AUDIT_READ, requested_amount as u64),
187            AuditControlOperation::IpcStat => (AUDIT_OP_IPC_STAT, 0),
188        };
189
190        Self {
191            sequence: 0,
192            timestamp_ms,
193            operation,
194            errno: error_to_errno(error),
195            subject_pid: subject.pid as u64,
196            subject_uid: subject.uid,
197            object_id: AUDIT_OBJECT_NONE,
198            object_owner_uid: AUDIT_UID_UNKNOWN,
199            requested_amount,
200            result_value: 0,
201        }
202    }
203
204    /// 转换成用户态 ABI 副本；覆盖标志仅影响副本，不修改内部事件。
205    pub(crate) const fn to_record(&self, gap_before: bool) -> AuditRecordV1 {
206        AuditRecordV1 {
207            abi_version: AUDIT_ABI_VERSION,
208            record_size: AUDIT_RECORD_V1_SIZE,
209            operation: self.operation,
210            flags: if gap_before {
211                AUDIT_RECORD_F_GAP_BEFORE
212            } else {
213                0
214            },
215            errno: self.errno,
216            subject_uid: self.subject_uid,
217            object_owner_uid: self.object_owner_uid,
218            reserved0: 0,
219            sequence: self.sequence,
220            timestamp_ms: self.timestamp_ms,
221            subject_pid: self.subject_pid,
222            object_id: self.object_id,
223            requested_amount: self.requested_amount,
224            result_value: self.result_value,
225            reserved1: 0,
226        }
227    }
228}

230/// 不包含借用的有界事件快照。
231///
232/// 系统调用层取得快照后，再逐条转换 ABI 记录并复制到用户内存。
233#[derive(Clone, Copy)]
234pub(crate) struct AuditBatch {
235    pub(crate) events: [AuditEvent; AUDIT_READ_MAX_RECORDS],
236    pub(crate) len: usize,
237    pub(crate) gap_before: bool,
238}
239
240impl AuditBatch {
241    const fn empty() -> Self {
242        Self {
243            events: [AuditEvent::EMPTY; AUDIT_READ_MAX_RECORDS],
244            len: 0,
245            gap_before: false,
246        }
247    }
248
249    /// 只转换有效记录，覆盖标志最多设置在本批次的第一条记录上。
250    /// 逐条转换可避免在 8 KiB 内核栈上再放置一整批 ABI 数组。
251    pub(crate) fn record_at(&self, index: usize) -> Option<AuditRecordV1> {
252        if index >= self.len {
253            return None;
254        }
255        self.events
256            .get(index)
257            .map(|event| event.to_record(index == 0 && self.gap_before))
258    }
259}

294/// 固定容量的环形缓冲区，只保留最近的事件。
295struct AuditRing {
296    events: [AuditEvent; AUDIT_RING_CAPACITY],
297    /// 最旧事件所在的数组下标。
298    head: usize,
299    /// 当前有效事件数量。
300    len: usize,
301    /// 下一条接受写入的事件序号。
302    next_sequence: u64,
303    total_events: u64,
304    successful_events: u64,
305    failed_events: u64,
306    overwritten_events: u64,
307}
308
309impl AuditRing {
310    const fn new() -> Self {
311        Self {
312            events: [AuditEvent::EMPTY; AUDIT_RING_CAPACITY],
313            head: 0,
314            len: 0,
315            next_sequence: 1,
316            total_events: 0,
317            successful_events: 0,
318            failed_events: 0,
319            overwritten_events: 0,
320        }
321    }
322
323    /// 写入事件，缓冲区满时覆盖最旧事件。
324    ///
325    /// 序号耗尽时停止接受新事件，避免回绕和重复使用序号。
326    /// 无论审计是否写入，调用方的 IPC 操作结果都不能改变。
327    fn push(&mut self, mut event: AuditEvent) {
328        let Some(next_sequence) = self.next_sequence.checked_add(1) else {
329            return;
330        };
331        event.sequence = self.next_sequence;
332        self.next_sequence = next_sequence;
333        // 未满使用空槽，满了就覆盖最老的事件。
334        if self.len < AUDIT_RING_CAPACITY {
335            let index = (self.head + self.len) % AUDIT_RING_CAPACITY;
336            self.events[index] = event;
337            self.len += 1;
338        } else {
339            self.events[self.head] = event;
340            self.head = (self.head + 1) % AUDIT_RING_CAPACITY;
341            self.overwritten_events += 1;
342        }
343
344        self.total_events += 1;
345        if event.errno == 0 {
346            self.successful_events += 1;
347        } else {
348            self.failed_events += 1;
349        }
350    }
351
352    /// 返回最旧序号，空缓冲区返回下一条序号。
353    fn first_sequence(&self) -> u64 {
354        if self.len == 0 {
355            self.next_sequence
356        } else {
357            self.events[self.head].sequence
358        }
359    }
360
361    /// 按游标生成非破坏性快照，读取不会删除记录或修改统计。
362    fn snapshot(&self, after_sequence: u64, requested_capacity: usize) -> AuditBatch {
363        let limit = requested_capacity.min(AUDIT_READ_MAX_RECORDS);
364        if limit == 0 || self.len == 0 || after_sequence == u64::MAX {
365            return AuditBatch::empty();
366        }
367
368        // 上面已排除 u64::MAX，因此加 1 不会溢出。
369        let wanted_sequence = after_sequence + 1;
370        let first_sequence = self.first_sequence();
371        let start_sequence = max(wanted_sequence, first_sequence);
372
373        if start_sequence >= self.next_sequence {
374            return AuditBatch::empty();
375        }
376
377        let count = (self.next_sequence - start_sequence).min(limit as u64) as usize;
378        let offset = (start_sequence - first_sequence) as usize;
379        let mut batch = AuditBatch::empty();
380
381        for batch_index in 0..count {
382            let ring_index = (self.head + offset + batch_index) % AUDIT_RING_CAPACITY;
383            batch.events[batch_index] = self.events[ring_index];
384        }
385
386        batch.len = count;
387        batch.gap_before = wanted_sequence < first_sequence;
388        batch
389    }
390
391    /// 在一次借用中复制全部统计，保持字段间的不变量。
392    fn stats(&self) -> AuditStatsSnapshot {
393        AuditStatsSnapshot {
394            capacity: AUDIT_RING_CAPACITY as u64,
395            retained: self.len as u64,
396            first_sequence: self.first_sequence(),
397            next_sequence: self.next_sequence,
398            total_events: self.total_events,
399            successful_events: self.successful_events,
400            failed_events: self.failed_events,
401            overwritten_events: self.overwritten_events,
402        }
403    }
404}

406lazy_static! {
407    /// 当前单核内核的全局审计状态；借用期间禁止调度或访问用户内存。
408    static ref AUDIT_RING: UPSafeCell<AuditRing> =
409        unsafe { UPSafeCell::new(AuditRing::new()) };
410}

412/// 返回值快照；函数返回时审计缓冲区的借用已经释放。
413pub(crate) fn snapshot(after_sequence: u64, requested_capacity: usize) -> AuditBatch {
414    let ring = AUDIT_RING.exclusive_access();
415    ring.snapshot(after_sequence, requested_capacity)
416}

418/// 返回统计值快照，不记录查询自身。
419pub(crate) fn stats() -> AuditStatsSnapshot {
420    let ring = AUDIT_RING.exclusive_access();
421    ring.stats()
422}

424/// 记录 IPC 结果，不修改或替换调用方持有的原始结果。
425///
426/// 时钟读取和事件转换都在借用环形缓冲区之前完成。
427pub(crate) fn record(request: &IpcRequest, outcome: &IpcResult<usize>) {
428    // 成功读取不生成新记录，避免读取者永远追不上日志尾部。
429    if matches!(request.operation, IpcOperation::AuditRead) && outcome.is_ok() {
430        return;
431    }
432
433    let timestamp_ms = crate::timer::get_time_ms() as u64;
434    let event = AuditEvent::from_request(request, outcome, timestamp_ms);
435    let mut ring = AUDIT_RING.exclusive_access();
436    ring.push(event);
437}

439/// 审计读取/统计的失败专用入口；成功的控制调用不应调用此函数。
440/// 同一次失败只选择一个记录入口，避免与 record 重复记录。
441pub(crate) fn record_control_failure(
442    operation: AuditControlOperation,
443    subject: &IpcSubject,
444    requested_amount: usize,
445    error: IpcError,
446) {
447    let timestamp_ms = crate::timer::get_time_ms() as u64;
448    let event =
449        AuditEvent::from_control_failure(operation, subject, requested_amount, error, timestamp_ms);
450    let mut ring = AUDIT_RING.exclusive_access();
451    ring.push(event);
452}
