17/// Security-related state embedded once in each process.
18pub struct ProcessSecurityState {
19    pub credentials: Credentials,
20    pub quota: QuotaState,
21}
22
23impl ProcessSecurityState {
24    /// Construct state for the initial process.
25    pub const fn initial() -> Self {
26        Self {
27            credentials: Credentials::initial(),
28            quota: QuotaState::initial(),
29        }
30    }
31
32    /// Construct child state through module-owned inheritance hooks.
33    pub const fn fork_from(parent: &Self) -> Self {
34        Self {
35            credentials: Credentials::fork_from(&parent.credentials),
36            quota: QuotaState::fork_from(&parent.quota),
37        }
38    }
39}

41/// Opaque proof that policy and quota checks ran for a request.
42pub struct IpcPermit {
43    request: IpcRequest,
44    reservation: QuotaReservation,
45}
46
47/// Run authorization and quota checks through stable module boundaries.
48pub fn preflight(state: &mut ProcessSecurityState, request: IpcRequest) -> IpcResult<IpcPermit> {
49    if let Err(error) = policy::authorize(&request) {
50        audit::record(&request, &Err(error));
51        return Err(error);
52    }
53
54    let reservation = match quota::reserve(&mut state.quota, &request) {
55        Ok(reservation) => reservation,
56        Err(error) => {
57            audit::record(&request, &Err(error));
58            return Err(error);
59        }
60    };
61
62    Ok(IpcPermit {
63        request,
64        reservation,
65    })
66}
67
68/// Finish a request, update quota state, and emit its audit outcome.
69pub fn complete(
70    state: &mut ProcessSecurityState,
71    permit: IpcPermit,
72    outcome: IpcResult<usize>,
73) -> IpcResult<usize> {
74    quota::finish(&mut state.quota, permit.reservation, outcome.is_ok());
75    audit::record(&permit.request, &outcome);
76    outcome
77}
78
79/// Record a request rejected before a permit can be issued (for example EFAULT
80/// found while validating a pipe buffer). No quota reservation is involved.
81pub fn record_failure(request: &IpcRequest, error: IpcError) {
82    audit::record(request, &Err(error));
83}
84/// Reserve one ordinary file-descriptor slot.
85///
86/// This crate-private hook keeps syscall code outside the quota module.
87pub(crate) fn reserve_file_fd(state: &mut ProcessSecurityState) -> IpcResult<()> {
88    state.quota.reserve_files(1)
89}
90
91/// Reserve quota for duplicating an existing descriptor.
92///
93/// Returns whether the source descriptor is a pipe endpoint.
94pub(crate) fn reserve_dup_fd(
95    state: &mut ProcessSecurityState,
96    source_fd: usize,
97) -> IpcResult<bool> {
98    state.quota.reserve_dup_fd(source_fd)
99}
100
101/// Register an already-reserved descriptor as a pipe endpoint.
102pub(crate) fn register_pipe_fd(state: &mut ProcessSecurityState, fd: usize) {
103    state.quota.register_pipe_fd(fd);
104}
105
106/// Remove pipe metadata without releasing the pending quota reservation.
107pub(crate) fn unregister_pipe_fd(state: &mut ProcessSecurityState, fd: usize) {
108    state.quota.unregister_pipe_fd(fd);
109}
110
111/// Release one committed descriptor and its associated quota.
112pub(crate) fn release_fd(state: &mut ProcessSecurityState, fd: usize) {
113    state.quota.release_fd(fd);
