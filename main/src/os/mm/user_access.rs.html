1//! Compatibility boundary for safe user-memory access.
2//!
3//! All user pointers are validated before dereference; a bad pointer
4//! yields `IpcError::InvalidAddress` instead of panicking the kernel.
5
6use core::mem::{MaybeUninit, size_of};
7
8use super::{UserBuffer, try_translated_byte_buffer};
9use crate::security::{IpcError, IpcResult};
10
11/// Copy a plain value from user memory.
12pub fn copy_from_user<T: Copy>(token: usize, src: *const T) -> IpcResult<T> {
13    let size = size_of::<T>();
14    if size == 0 {
15        return Err(IpcError::InvalidArgument);
16    }
17
18    let buffers = try_translated_byte_buffer(token, src.cast::<u8>(), size, false)
19        .ok_or(IpcError::InvalidAddress)?;
20    let mut value = MaybeUninit::<T>::uninit();
21    let destination =
22        unsafe { core::slice::from_raw_parts_mut(value.as_mut_ptr().cast::<u8>(), size) };
23    let mut offset = 0;
24    for buffer in buffers {
25        let end = offset + buffer.len();
26        destination[offset..end].copy_from_slice(buffer);
27        offset = end;
28    }
29    debug_assert_eq!(offset, size);
30    Ok(unsafe { value.assume_init() })
31}
32
33/// Copy a plain value to user memory.
34pub fn copy_to_user<T: Copy>(token: usize, dst: *mut T, value: &T) -> IpcResult<()> {
35    let size = size_of::<T>();
36    if size == 0 {
37        return Err(IpcError::InvalidArgument);
38    }
39
40    let buffers = try_translated_byte_buffer(token, dst.cast::<u8>(), size, true)
41        .ok_or(IpcError::InvalidAddress)?;
42    let source = unsafe { core::slice::from_raw_parts((value as *const T).cast::<u8>(), size) };
43    let mut offset = 0;
44    for buffer in buffers {
45        let end = offset + buffer.len();
46        buffer.copy_from_slice(&source[offset..end]);
47        offset = end;
48    }
49    debug_assert_eq!(offset, size);
50    Ok(())
51}
52
53/// Validate bytes that the kernel will read and return a page-segmented buffer.
54pub fn copy_bytes_from_user(token: usize, src: *const u8, len: usize) -> IpcResult<UserBuffer> {
55    let buffers =
56        try_translated_byte_buffer(token, src, len, false).ok_or(IpcError::InvalidAddress)?;
57    Ok(UserBuffer::new(buffers))
58}
