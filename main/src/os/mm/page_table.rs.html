18/// SV39 用户虚拟地址空间正区上限（2^38）。
19///
20/// 用户指针及其覆盖区间必须严格位于 `[0, USER_VA_LIMIT)` 内，
21/// 超出该范围的地址（内核区、trampoline、trap context 等）一律拒绝。
22const USER_VA_LIMIT: usize = 1 << 38;

24bitflags! {
25    pub struct PTEFlags: u8 {
26        const V = 1 << 0;
27        const R = 1 << 1;
28        const W = 1 << 2;
29        const X = 1 << 3;
30        const U = 1 << 4;
31        const G = 1 << 5;
32        const A = 1 << 6;
33        const D = 1 << 7;
34    }
35}

37#[derive(Copy, Clone)]
38#[repr(C)]
39pub struct PageTableEntry {
40    pub bits: usize,
41}
42
43impl PageTableEntry {
44    pub fn new(ppn: PhysPageNum, flags: PTEFlags) -> Self {
45        PageTableEntry {
46            bits: ppn.0 << 10 | flags.bits as usize,
47        }
48    }
49    pub fn empty() -> Self {
50        PageTableEntry { bits: 0 }
51    }
52    pub fn ppn(&self) -> PhysPageNum {
53        (self.bits >> 10 & ((1usize << 44) - 1)).into()
54    }
55    pub fn flags(&self) -> PTEFlags {
56        PTEFlags::from_bits(self.bits as u8).unwrap()
57    }
58    pub fn is_valid(&self) -> bool {
59        (self.flags() & PTEFlags::V) != PTEFlags::empty()
60    }
61    pub fn readable(&self) -> bool {
62        (self.flags() & PTEFlags::R) != PTEFlags::empty()
63    }
64    pub fn writable(&self) -> bool {
65        (self.flags() & PTEFlags::W) != PTEFlags::empty()
66    }
67    pub fn executable(&self) -> bool {
68        (self.flags() & PTEFlags::X) != PTEFlags::empty()
69    }
70}

72pub struct PageTable {
73    root_ppn: PhysPageNum,
74    frames: Vec<FrameTracker>,
75}
76
77/// Assume that it won't oom when creating/mapping.
78impl PageTable {
79    pub fn new() -> Self {
80        let frame = frame_alloc().unwrap();
81        PageTable {
82            root_ppn: frame.ppn,
83            frames: vec![frame],
84        }
85    }
86    /// Temporarily used to get arguments from user space.
87    pub fn from_token(satp: usize) -> Self {
88        Self {
89            root_ppn: PhysPageNum::from(satp & ((1usize << 44) - 1)),
90            frames: Vec::new(),
91        }
92    }
93    fn find_pte_create(&mut self, vpn: VirtPageNum) -> Option<&mut PageTableEntry> {
94        let idxs = vpn.indexes();
95        let mut ppn = self.root_ppn;
96        let mut result: Option<&mut PageTableEntry> = None;
97        for (i, idx) in idxs.iter().enumerate() {
98            let pte = &mut ppn.get_pte_array()[*idx];
99            if i == 2 {
100                result = Some(pte);
101                break;
102            }
103            if !pte.is_valid() {
104                let frame = frame_alloc().unwrap();
105                *pte = PageTableEntry::new(frame.ppn, PTEFlags::V);
106                self.frames.push(frame);
107            }
108            ppn = pte.ppn();
109        }
110        result
111    }
112    fn find_pte(&self, vpn: VirtPageNum) -> Option<&mut PageTableEntry> {
113        let idxs = vpn.indexes();
114        let mut ppn = self.root_ppn;
115        let mut result: Option<&mut PageTableEntry> = None;
116        for (i, idx) in idxs.iter().enumerate() {
117            let pte = &mut ppn.get_pte_array()[*idx];
118            if i == 2 {
119                result = Some(pte);
120                break;
121            }
122            if !pte.is_valid() {
123                return None;
124            }
125            ppn = pte.ppn();
126        }
127        result
128    }
129    #[allow(unused)]
130    pub fn map(&mut self, vpn: VirtPageNum, ppn: PhysPageNum, flags: PTEFlags) {
131        let pte = self.find_pte_create(vpn).unwrap();
132        assert!(!pte.is_valid(), "vpn {:?} is mapped before mapping", vpn);
133        *pte = PageTableEntry::new(ppn, flags | PTEFlags::V);
134    }
135    #[allow(unused)]
136    pub fn unmap(&mut self, vpn: VirtPageNum) {
137        let pte = self.find_pte(vpn).unwrap();
138        assert!(pte.is_valid(), "vpn {:?} is invalid before unmapping", vpn);
139        *pte = PageTableEntry::empty();
140    }
141    pub fn translate(&self, vpn: VirtPageNum) -> Option<PageTableEntry> {
142        self.find_pte(vpn).map(|pte| *pte)
143    }
144    pub fn translate_va(&self, va: VirtAddr) -> Option<PhysAddr> {
145        self.find_pte(va.clone().floor()).map(|pte| {
146            let aligned_pa: PhysAddr = pte.ppn().into();
147            let offset = va.page_offset();
148            let aligned_pa_usize: usize = aligned_pa.into();
149            (aligned_pa_usize + offset).into()
150        })
151    }
152    pub fn token(&self) -> usize {
153        8usize << 60 | self.root_ppn.0
154    }
155}

157/// 校验用户提供的地址区间 `[ptr, ptr + len)` 是否安全可访问。
158///
159/// 通过要求：
160/// 1. `ptr + len` 不产生 usize 溢出，且整个区间落在用户正区 `[0, USER_VA_LIMIT)` 内；
161/// 2. 区间覆盖的每一页在当前用户页表中均已映射；
162/// 3. 每一页的 PTE 均带 `U` 标志（拒绝内核页 / trap context / trampoline）；
163/// 4. 每一页的权限满足访问意图（`write == true` 要求 `W`，否则要求 `R`）。
164///
165/// 返回 `false` 表示该区间不可访问，调用方应拒绝本次访问（如返回 -1），
166/// 绝不能继续解引用。
167pub fn check_user_range(token: usize, ptr: usize, len: usize, write: bool) -> bool {
168    if len == 0 {
169        return true;
170    }
171    let Some(end) = ptr.checked_add(len) else {
172        return false;
173    };
174    if ptr >= USER_VA_LIMIT || end > USER_VA_LIMIT {
175        return false;
176    }
177    let page_table = PageTable::from_token(token);
178    let mut cur = ptr;
179    while cur < end {
180        let pte = match page_table.translate(VirtAddr::from(cur).floor()) {
181            Some(pte) => pte,
182            None => return false,
183        };
184        let flags = pte.flags();
185        if !flags.contains(PTEFlags::U) {
186            return false;
187        }
188        if write {
189            if !flags.contains(PTEFlags::W) {
190                return false;
191            }
192        } else if !flags.contains(PTEFlags::R) {
193            return false;
194        }
195        // 跳到下一页边界（或区间终点）
196        let page_end = (cur & !(PAGE_SIZE - 1)) + PAGE_SIZE;
197        cur = page_end.min(end);
198    }
199    true
200}

202/// 安全版 `translated_byte_buffer`。
203///
204/// `write` 表示内核的访问意图：内核将向该缓冲区写入（如 `read` 系统调用）
205/// 时传 `true`，要求用户页可写；内核将从该缓冲区读取（如 `write` 系统调用）
206/// 时传 `false`，要求用户页可读。区间非法时返回 `None`，调用方应优雅失败。
207pub fn try_translated_byte_buffer(
208    token: usize,
209    ptr: *const u8,
210    len: usize,
211    write: bool,
212) -> Option<Vec<&'static mut [u8]>> {
213    if !check_user_range(token, ptr as usize, len, write) {
214        return None;
215    }
216    // 校验通过后，以下逐页翻译不会触发未映射 panic
217    let page_table = PageTable::from_token(token);
218    let mut start = ptr as usize;
219    let end = start + len;
220    let mut v = Vec::new();
221    while start < end {
222        let start_va = VirtAddr::from(start);
223        let mut vpn = start_va.floor();
224        let ppn = page_table.translate(vpn).unwrap().ppn();
225        vpn.step();
226        let mut end_va: VirtAddr = vpn.into();
227        end_va = end_va.min(VirtAddr::from(end));
228        if end_va.page_offset() == 0 {
229            v.push(&mut ppn.get_bytes_array()[start_va.page_offset()..]);
230        } else {
231            v.push(&mut ppn.get_bytes_array()[start_va.page_offset()..end_va.page_offset()]);
232        }
233        start = end_va.into();
234    }
235    Some(v)
236}

238/// 安全版 `translated_str`：从用户空间加载以 `\0` 结尾的字符串。
239///
240/// 逐页扫描：只要求“实际扫描过的页”已映射、带 `U` 标志、可读，
241/// 一旦遇到 `\0` 立即成功返回；因此字符串很短时不会要求其后的内存也映射。
242/// 若扫描超过 `max_len` 仍无终止符（或中途遇到非法页），返回 `None`。
243pub fn try_translated_str(token: usize, ptr: *const u8, max_len: usize) -> Option<String> {
244    let start = ptr as usize;
245    if start == 0 || max_len == 0 {
246        return None;
247    }
248    // 仅做数值边界检查：扫描上限不得溢出、不得越过用户正区
249    let limit = start.checked_add(max_len)?;
250    if limit > USER_VA_LIMIT {
251        return None;
252    }
253    let page_table = PageTable::from_token(token);
254    let mut string = String::new();
255    let mut cur = start;
256    while cur < limit {
257        // 只校验当前扫描到的这一页：必须映射、带 U 标志、可读
258        let pte = page_table.translate(VirtAddr::from(cur).floor())?;
259        let flags = pte.flags();
260        if !flags.contains(PTEFlags::U) || !flags.contains(PTEFlags::R) {
261            return None;
262        }
263        let ppn = pte.ppn();
264        let offset = cur & (PAGE_SIZE - 1);
265        let chunk = (PAGE_SIZE - offset).min(limit - cur);
266        let bytes = ppn.get_bytes_array();
267        let slice = &bytes[offset..offset + chunk];
268        match slice.iter().position(|&b| b == 0) {
269            Some(z) => {
270                string.extend(slice[..z].iter().map(|&b| b as char));
271                return Some(string);
272            }
273            None => {
274                string.extend(slice.iter().map(|&b| b as char));
275                cur += chunk;
276            }
277        }
278    }
279    None
280}

282/// 安全版 `translated_ref`：校验后返回对用户内存中 `T` 的共享引用。
283///
284/// 要求 `T` 完整落在同一用户页内（避免跨页时物理内存不连续导致读错数据），
285/// 且该页已映射、带 `U` 标志、可读。非法时返回 `None`。
286pub fn try_translated_ref<T>(token: usize, ptr: *const T) -> Option<&'static T> {
287    let addr = ptr as usize;
288    let size = core::mem::size_of::<T>();
289    if size == 0 || addr == 0 || addr % core::mem::align_of::<T>() != 0 {
290        return None;
291    }
292    let end = addr.checked_add(size)?;
293    if end > USER_VA_LIMIT {
294        return None;
295    }
296    // 跨页检查：整个 T 必须位于同一页内
297    if (addr >> PAGE_SIZE_BITS) != ((end - 1) >> PAGE_SIZE_BITS) {
298        return None;
299    }
300    if !check_user_range(token, addr, size, false) {
301        return None;
302    }
303    let page_table = PageTable::from_token(token);
304    page_table
305        .translate_va(VirtAddr::from(addr))
306        .map(|pa| pa.get_ref())
307}

364pub struct UserBufferIterator {
365    buffers: Vec<&'static mut [u8]>,
366    current_buffer: usize,
367    current_idx: usize,
368}
369
370impl Iterator for UserBufferIterator {
371    type Item = *mut u8;
372    fn next(&mut self) -> Option<Self::Item> {
373        if self.current_buffer >= self.buffers.len() {
374            None
375        } else {
376            let r = &mut self.buffers[self.current_buffer][self.current_idx] as *mut _;
377            if self.current_idx + 1 == self.buffers[self.current_buffer].len() {
378                self.current_idx = 0;
379                self.current_buffer += 1;
380            } else {
381                self.current_idx += 1;
382            }
383            Some(r)
384        }
385    }
