1//! 审计系统调用主体：602 读取事件，603 查询统计。
2//!
3//! 调用顺序为“复制调用者身份 → 检查权限/参数 → 取得值快照 → 复制用户内存”。
4//! 任务状态和审计缓冲区的借用都不能跨越用户内存复制；成功查询不记录自身。
5//! 页映射、可写权限和跨页复制由 mm::copy_to_user 保证，需配合 B 的加固实现。
6
7use core::mem::size_of;
8
9use crate::mm::copy_to_user;
10use crate::security::audit::{self, AuditControlOperation, AuditRecordV1, IpcStatsV1};
11use crate::security::{CapabilitySet, IpcError, IpcResult, IpcSubject};
12use crate::task::current_task;

14/// 从当前任务复制身份和地址空间标识，不把 RefMut 带到审计或复制路径中。
15fn caller_context() -> (IpcSubject, usize) {
16    // 系统调用只能由正在运行的用户任务进入；缺少任务属于内核调度错误。
17    let task = current_task().expect("审计系统调用必须存在当前任务");
18    let inner = task.inner_exclusive_access();
19    let credentials = inner.security.credentials;
20    let subject = IpcSubject {
21        pid: task.getpid(),
22        uid: credentials.uid,
23        capabilities: credentials.capabilities,
24    };
25    let token = inner.get_user_token();
26    // 返回的都是普通值；函数返回前自动释放任务内部借用。
27    (subject, token)
28}

30/// 两个调用使用同一权限规则；先检查身份，再处理参数或生成快照。
31fn check_permission(subject: &IpcSubject) -> IpcResult<()> {
32    if subject.uid == 0 || subject.capabilities.contains(CapabilitySet::AUDIT_READ) {
33        Ok(())
34    } else {
35        Err(IpcError::PermissionDenied)
36    }
37}

39/// 仅检查实际输出范围的空地址与整数溢出，不代替用户页表检查。
40fn check_output_range(start: usize, count: usize, item_size: usize) -> IpcResult<()> {
41    let bytes = count
42        .checked_mul(item_size)
43        .ok_or(IpcError::InvalidAddress)?;
44    if bytes != 0 && start == 0 {
45        return Err(IpcError::InvalidAddress);
46    }
47    start.checked_add(bytes).ok_or(IpcError::InvalidAddress)?;
48    Ok(())
49}

51/// 统一控制调用的出口：失败只记一次，日志保存正 errno，用户收到负 errno。
52fn finish_control_call(
53    operation: AuditControlOperation,
54    subject: &IpcSubject,
55    requested_amount: usize,
56    outcome: IpcResult<usize>,
57) -> isize {
58    match outcome {
59        // 成功只可能是 0 或不超过 32 的记录数，不调用 complete，避免自反馈。
60        Ok(value) => value as isize,
61        Err(error) => {
62            audit::record_control_failure(operation, subject, requested_amount, error);
63            -(audit::error_to_errno(error) as isize)
64        }
65    }
66}

68/// 系统调用 602：返回 sequence > after_sequence 的保留记录，最多 32 条。
69///
70/// records 是用户地址，只能交给 copy_to_user；内核不能直接解引用它。
71/// 返回负值时，用户缓冲区可能已有前缀数据，用户必须保留原游标重试。
72pub(crate) fn sys_audit_read(
73    records: *mut AuditRecordV1,
74    capacity: usize,
75    after_sequence: u64,
76) -> isize {
77    let (subject, token) = caller_context();
78    let outcome = read_records(&subject, token, records, capacity, after_sequence);
79    finish_control_call(
80        AuditControlOperation::AuditRead,
81        &subject,
82        capacity,
83        outcome,
84    )
85}

87fn read_records(
88    subject: &IpcSubject,
89    token: usize,
90    records: *mut AuditRecordV1,
91    capacity: usize,
92    after_sequence: u64,
93) -> IpcResult<usize> {
94    check_permission(subject)?;
95    if capacity == 0 {
96        // 授权调用者可以传空指针；零容量不生成快照，也不访问输出地址。
97        return Ok(0);
98    }
99
100    // snapshot 内部将容量截断到 32。它返回独立值，返回时审计借用已经释放。
101    // 后续复制期间追加的事件不会混入本批次。
102    let batch = audit::snapshot(after_sequence, capacity);
103    if batch.len == 0 {
104        return Ok(0);
105    }
106
107    let start = records as usize;
108    let record_size = size_of::<AuditRecordV1>();
109    // 只检查实际返回的记录范围；capacity 很大并不要求分配或校验整个用户数组。
110    check_output_range(start, batch.len, record_size)?;
111
112    for index in 0..batch.len {
113        // 逐条构造 80 字节的 ABI 副本，不再在 8 KiB 内核栈上创建第二个批次数组。
114        let record = batch.record_at(index).ok_or(IpcError::InvalidArgument)?;
115        // 整段范围已经通过 checked_mul/checked_add，以下偏移不会回绕。
116        let destination = (start + index * record_size) as *mut AuditRecordV1;
117        copy_to_user(token, destination, &record).map_err(|_| IpcError::InvalidAddress)?;
118    }
119
120    // 读取不删除日志，也不在内核保存游标。全部复制成功后才报告记录数。
121    Ok(batch.len)
122}

124/// 系统调用 603：返回一个一致的 80 字节统计快照，成功返回 0。
125///
126/// v1 只接受 flags == 0 和 out_size >= 80；多出的尾部字节保持不变。
127pub(crate) fn sys_ipc_stat(stats: *mut IpcStatsV1, out_size: usize, flags: usize) -> isize {
128    let (subject, token) = caller_context();
129    let outcome = read_stats(&subject, token, stats, out_size, flags);
130    finish_control_call(AuditControlOperation::IpcStat, &subject, 0, outcome)
131}
132
133fn read_stats(
134    subject: &IpcSubject,
135    token: usize,
136    stats: *mut IpcStatsV1,
137    out_size: usize,
138    flags: usize,
139) -> IpcResult<usize> {
140    check_permission(subject)?;
141    let struct_size = size_of::<IpcStatsV1>();
142    if flags != 0 || out_size < struct_size {
143        // 参数错误优先于地址检查，非法 flags 不触碰用户指针。
144        return Err(IpcError::InvalidArgument);
145    }
146    check_output_range(stats as usize, 1, struct_size)?;
147
148    // 全部计数来自一次借用。转换和用户复制都在审计借用释放之后进行。
149    let record = audit::stats().to_record();
150    copy_to_user(token, stats, &record).map_err(|_| IpcError::InvalidAddress)?;
151    Ok(0)
152}
