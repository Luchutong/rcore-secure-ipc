15fn ipc_error_to_ret(error: IpcError) -> isize {
16    match error {
17        IpcError::PermissionDenied => -1,
18        IpcError::InvalidAddress => -14,
19        IpcError::InvalidArgument => -22,
20        IpcError::ProcessNotFound => -3,
21        IpcError::TooManyFiles => -24,
22        IpcError::ResourceExhausted => -28,
23        IpcError::TryAgain => -11,
24    }
25}

45pub fn sys_fork() -> isize {
46    let current_task = current_task().unwrap();
47    let new_task = current_task.fork();
48    let new_pid = new_task.pid.0;
49    // modify trap context of new_task, because it returns immediately after switching
50    let trap_cx = new_task.inner_exclusive_access().get_trap_cx();
51    // we do not have to move to next instruction since we have done it before
52    // for child process, fork returns 0
53    trap_cx.x[10] = 0;
54    // add new task to scheduler
55    add_task(new_task);
56    new_pid as isize
57}

59pub fn sys_exec(path: *const u8, mut args: *const usize) -> isize {
60    let token = current_user_token();
61    // 校验路径字符串：非法指针或超长返回 -1
62    let Some(path) = try_translated_str(token, path, MAX_STR_LEN) else {
63        return -14;
64    };
65    let mut args_vec: Vec<String> = Vec::new();
66    loop {
67        // 逐项校验参数指针：非法地址返回 -1，不 panic
68        let Ok(arg_str_ptr) = copy_from_user(token, args) else {
69            return -14;
70        };
71        if arg_str_ptr == 0 {
72            break;
73        }
74        // 限制参数个数，防止未以 0 结尾的 argv 数组导致无限遍历
75        if args_vec.len() >= MAX_ARGV {
76            return -1;
77        }
78        let Some(arg) = try_translated_str(token, arg_str_ptr as *const u8, MAX_STR_LEN) else {
79            return -14;
80        };
81        args_vec.push(arg);
82        args = args.wrapping_add(1);
83    }
84    if let Some(app_inode) = open_file(path.as_str(), OpenFlags::RDONLY) {
85        let all_data = app_inode.read_all();
86        let task = current_task().unwrap();
87        let argc = args_vec.len();
88        task.exec(all_data.as_slice(), args_vec);
89        // return argc because cx.x[10] will be covered with it later
90        argc as isize
91    } else {
92        -1
93    }
94}

96/// If there is not a child process whose pid is same as given, return -1.
97/// Else if there is a child process but it is still running, return -2.
98pub fn sys_waitpid(pid: isize, exit_code_ptr: *mut i32) -> isize {
99    let task = current_task().unwrap();
100    // find a child process
101
102    // ---- access current PCB exclusively
103    let mut inner = task.inner_exclusive_access();
104    if !inner
105        .children
106        .iter()
107        .any(|p| pid == -1 || pid as usize == p.getpid())
108    {
109        return -1;
110        // ---- release current PCB
111    }
112    let pair = inner.children.iter().enumerate().find(|(_, p)| {
113        // ++++ temporarily access child PCB exclusively
114        p.inner_exclusive_access().is_zombie() && (pid == -1 || pid as usize == p.getpid())
115        // ++++ release child PCB
116    });
117    if let Some((idx, _)) = pair {
118        let child = &inner.children[idx];
119        let found_pid = child.getpid();
120        // ++++ temporarily access child PCB exclusively
121        let exit_code = child.inner_exclusive_access().exit_code;
122        // ++++ release child PCB
123        // 先复制退出码：非法地址返回 EFAULT，且不移动 children 列表，调用方可重试。
124        if copy_to_user(inner.memory_set.token(), exit_code_ptr, &exit_code).is_err() {
125            return -14;
126        }
127        let child = inner.children.remove(idx);
128        // confirm that child will be deallocated after being removed from children list
129        assert_eq!(Arc::strong_count(&child), 1);
130        found_pid as isize
131    } else {
132        -2
133    }
134    // ---- release current PCB automatically
135}

137pub fn sys_kill(pid: usize, signum: i32) -> isize {
138    if signum < 0 || signum as usize > MAX_SIG {
139        return ipc_error_to_ret(IpcError::InvalidArgument);
140    }
141    let flag = match SignalFlags::from_bits(1u32 << signum as u32) {
142        Some(f) => f,
143        None => return ipc_error_to_ret(IpcError::InvalidArgument),
144    };
145
146    let target = match pid2task(pid) {
147        Some(t) => t,
148        None => return ipc_error_to_ret(IpcError::ProcessNotFound),
149    };
150
151    let caller = current_task().unwrap();
152    let caller_pid = caller.getpid();
153    let target_uid = {
154        let inner = target.inner_exclusive_access();
155        inner.security.credentials.uid
156    };
157
158    // A+D integration point: policy, quota bookkeeping and audit all pass
159    // through the stable facade. Authorization failures are recorded by preflight.
160    let permit = {
161        let mut inner = caller.inner_exclusive_access();
162        let credentials = inner.security.credentials;
163        let request = IpcRequest {
164            subject: IpcSubject {
165                pid: caller_pid,
166                uid: credentials.uid,
167                capabilities: credentials.capabilities,
168            },
169            object: IpcObject {
170                id: pid as u64,
171                owner_uid: target_uid,
172            },
173            operation: IpcOperation::SignalSend,
174            amount: 1,
175        };
176        match security::preflight(&mut inner.security, request) {
177            Ok(permit) => permit,
178            Err(error) => return ipc_error_to_ret(error),
179        }
180    };
181
182    let outcome = {
183        let mut target_inner = target.inner_exclusive_access();
184        if target_inner.signals.contains(flag) {
185            Err(IpcError::TryAgain)
186        } else {
187            target_inner.signals.insert(flag);
188            Ok(1)
189        }
190    };
191
192    let mut inner = caller.inner_exclusive_access();
193    match security::complete(&mut inner.security, permit, outcome) {
194        Ok(_) => 0,
195        Err(error) => ipc_error_to_ret(error),
196    }
197}

199pub fn sys_sigprocmask(mask: u32) -> isize {
200    if let Some(task) = current_task() {
201        let mut inner = task.inner_exclusive_access();
202        let old_mask = inner.signal_mask;
203        if let Some(flag) = SignalFlags::from_bits(mask) {
204            inner.signal_mask = flag;
205            old_mask.bits() as isize
206        } else {
207            -1
208        }
209    } else {
210        -1
211    }
212}

214pub fn sys_sigreturn() -> isize {
215    if let Some(task) = current_task() {
216        let mut inner = task.inner_exclusive_access();
217        inner.handling_sig = -1;
218        // restore the trap context
219        let trap_ctx = inner.get_trap_cx();
220        *trap_ctx = inner.trap_ctx_backup.unwrap();
221        // Here we return the value of a0 in the trap_ctx,
222        // otherwise it will be overwritten after we trap
223        // back to the original execution of the application.
224        trap_ctx.x[10] as isize
225    } else {
226        -1
227    }
228}

230fn check_sigaction_error(signal: SignalFlags, action: usize, old_action: usize) -> bool {
231    if action == 0
232        || old_action == 0
233        || signal == SignalFlags::SIGKILL
234        || signal == SignalFlags::SIGSTOP
235    {
236        true
237    } else {
238        false
239    }
240}

242pub fn sys_sigaction(
243    signum: i32,
244    action: *const SignalAction,
245    old_action: *mut SignalAction,
246) -> isize {
247    let token = current_user_token();
248    let task = current_task().unwrap();
249    let mut inner = task.inner_exclusive_access();
250    if signum as usize > MAX_SIG {
251        return -1;
252    }
253    if let Some(flag) = SignalFlags::from_bits(1 << signum) {
254        if check_sigaction_error(flag, action as usize, old_action as usize) {
255            return -1;
256        }
257        let prev_action = inner.signal_actions.table[signum as usize];
258        // 复制输入后再写回旧动作；任一地址非法都返回 EFAULT，不修改内核动作表。
259        let Ok(new_action) = copy_from_user(token, action) else {
260            return -14;
261        };
262        if copy_to_user(token, old_action, &prev_action).is_err() {
263            return -14;
264        }
265        inner.signal_actions.table[signum as usize] = new_action;
266        0
267    } else {
268        -1
269    }
270}

283/// Change the UID of the calling process.
284///
285/// Only root (UID 0) may call this.  After the call the process loses root
286/// privileges (its capabilities are cleared) unless `uid == 0`.
287/// Returns 0 on success, -1 on failure.
288pub fn sys_setuid(uid: usize) -> isize {
289    let task = current_task().unwrap();
290    let mut inner = task.inner_exclusive_access();
291    let cred = &mut inner.security.credentials;
292
293    // Only root can change UID.
294    if !cred.is_root() {
295        return ipc_error_to_ret(IpcError::PermissionDenied);
296    }
297
298    let Ok(new_uid) = Uid::try_from(uid) else {
299        return ipc_error_to_ret(IpcError::InvalidArgument);
300    };
301    cred.uid = new_uid;
302    // Dropping root: lose all capabilities.
303    // Staying root: keep all capabilities.
304    if new_uid != 0 {
305        cred.capabilities = CapabilitySet::empty();
306    }
307    0
308}
